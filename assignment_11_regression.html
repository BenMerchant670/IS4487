Original Source: Dataset Description¶

The dataset you'll be using is a detailed Airbnb listing file, available from Inside Airbnb.

Each row represents one property listing. The columns include:

  • Host attributes (e.g., host ID, host name, host response time)
  • Listing details (e.g., price, room type, minimum nights, availability)
  • Location data (e.g., neighborhood, latitude/longitude)
  • Property characteristics (e.g., number of bedrooms, amenities, accommodates)
  • Calendar/booking variables (e.g., last review date, number of reviews)

The schema is consistent across cities, so you can expect similar columns regardless of the location you choose.

from sklearn.linear_model import LinearRegression
from sklearn.impute import SimpleImputer
import pandas as pd
import numpy as np

# Remove identifiers, text, URLs, and other unsuitable columns
columns_to_remove = [
    'id', 'listing_url', 'scrape_id', 'host_id',
    'host_url', 'name', 'description', 'picture_url',
    'host_name', 'host_about', 'host_picture_url',
    'host_thumbnail_url', 'amenities', 'calendar_updated',
    'last_scraped', 'host_since', 'first_review',
    'last_review', 'calendar_last_scraped'
]

X_train_clean = X_train.drop(
    columns=columns_to_remove, errors='ignore'
).copy()

X_test_clean = X_test.drop(
    columns=columns_to_remove, errors='ignore'
).copy()

# Keep numeric features
X_train_clean = X_train_clean.select_dtypes(
    include='number'
)
X_test_clean = X_test_clean[X_train_clean.columns]

# Replace infinite values with missing values
X_train_clean = X_train_clean.replace(
    [np.inf, -np.inf], np.nan
)
X_test_clean = X_test_clean.replace(
    [np.inf, -np.inf], np.nan
)

# Remove columns containing only missing values
valid_columns = X_train_clean.columns[
    X_train_clean.notna().any()
]

X_train_clean = X_train_clean[valid_columns]
X_test_clean = X_test_clean[valid_columns]

# Fill missing values using training medians
imputer = SimpleImputer(strategy='median')

X_train_ready = pd.DataFrame(
    imputer.fit_transform(X_train_clean),
    columns=valid_columns,
    index=X_train_clean.index
)

X_test_ready = pd.DataFrame(
    imputer.transform(X_test_clean),
    columns=valid_columns,
    index=X_test_clean.index
)

# Remove rows with missing target prices
valid_train = y_train.notna()
valid_test = y_test.notna()

X_train_ready = X_train_ready.loc[valid_train]
X_test_ready = X_test_ready.loc[valid_test]

y_train_ready = y_train.loc[valid_train]
y_test_ready = y_test.loc[valid_test]

# Train linear regression
model = LinearRegression()
model.fit(X_train_ready, y_train_ready)

# Predict prices on the test data
y_pred = model.predict(X_test_ready)

print("Linear regression model trained successfully!")
print("Predictors used:", X_train_ready.shape[1])

# Preview actual vs predicted prices
results = pd.DataFrame({
    'Actual Price': y_test_ready,
    'Predicted Price': y_pred
})

display(results.head(10))
